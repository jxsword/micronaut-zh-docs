下面是一个用 Kotlin 编写的控制器示例：

[source, kotlin]
.src/main/kotlin/example/HelloController.kt
----
package example

import io.micronaut.http.annotation.*

@Controller("/")
class HelloController {

    @Get("/hello/{name}")
    fun hello(name: String): String {
        return "Hello $name"
    }
}
